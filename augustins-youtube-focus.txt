! ===================================================================
! AUGUSTIN'S YOUTUBE FOCUS FILTER
! Focus on what you search for, not what the algorithm feeds you.
! ===================================================================

! 1. Hide the Homepage Feed (Leaves a blank page with just the search bar)
youtube.com##ytd-browse[page-subtype="home"] ytd-rich-grid-renderer

! 2. Hide Watch Page Sidebar (Related / Up Next videos)
youtube.com##ytd-watch-flexy #related

! 3. Hide End Screen Video Suggestions (Pop-ups that appear when a video finishes)
youtube.com##.ytp-endscreen-content
youtube.com##.ytp-ce-element

! 4. Hide Shorts (Shelves and Sidebar Button)
youtube.com##ytd-reel-shelf-renderer
youtube.com##ytd-rich-section-renderer:has(yt-icon[icon="yt-sys-icons:shorts"])
youtube.com##[tab-identifier="FEshorts"]

! 5. Hide "Explore" and "Trending" sidebars 
youtube.com##ytd-guide-entry-renderer a[href^="/feed/explore"]
youtube.com##ytd-guide-entry-renderer a[href^="/feed/trending"]

! ===================================================================
! MOBILE WEB SUPPORT (For Firefox Android + uBlock Origin)
! ===================================================================
m.youtube.com##ytm-browse ytm-rich-grid-renderer
m.youtube.com##ytm-item-section-renderer[section-identifier="related-items"]
